                                                let restoring = false;

                                                function getCleanClone(template) {
                                                    const clone = template.cloneNode(true);
                                                    clone.classList.remove('hide');

                                                    clone.querySelectorAll('input, select, textarea').forEach(input => {
                                                        if (input.type === 'radio' || input.type === 'checkbox') {
                                                            input.checked = false;
                                                        } else {
                                                            input.value = '';
                                                        }
                                                        input.removeAttribute('required');
                                                    }
                                                    );

                                                    clone.querySelectorAll('label span').forEach(span => {
                                                        span.textContent = span.textContent.replace(/ \*$/, '');
                                                    }
                                                    );
                                                    clone.querySelectorAll('option').forEach(option => {
                                                        option.textContent = option.textContent.replace(/ \*$/, '');
                                                    }
                                                    );
                                                    clone.querySelectorAll('label').forEach(label => {
                                                        if (!label.querySelector('span')) {
                                                            label.textContent = label.textContent.replace(/ \*$/, '');
                                                        }
                                                    }
                                                    );

                                                    return clone;
                                                }

                                                const blankMainTemplate = getCleanClone(mainTemplate);
                                                const blankSecondaryTemplate = getCleanClone(secondaryTemplate);

                                                function updateNaming(input, newIndex) {
                                                    ['name', 'data-name'].forEach(attr => {
                                                        const original = input.getAttribute(attr);
                                                        if (!original)
                                                            return;
                                                        const match = original.match(/^child_(\d+)_(.+)$/);
                                                        if (match) {
                                                            const field = match[2];
                                                            input.setAttribute(attr, `child_${newIndex}_${field}`);
                                                        }
                                                    }
                                                    );

                                                    if (input.hasAttribute('id')) {
                                                        const originalId = input.getAttribute('id');
                                                        const base = originalId.replace(/-\d+$/, '');
                                                        input.setAttribute('id', `${base}-${newIndex}`);
                                                    }
                                                }

                                                function enforceChildLimit() {
                                                    const total = mainList.querySelectorAll('[sw-child-template="main"]').length + 1;
                                                    if (total >= 3) {
                                                        duplicateBtn.classList.add('is-disabled');
                                                        duplicateBtn.setAttribute('disabled', 'disabled');
                                                    } else {
                                                        duplicateBtn.classList.remove('is-disabled');
                                                        duplicateBtn.removeAttribute('disabled');
                                                    }
                                                }

                                                function updateIndexes() {
                                                    const mainItems = [mainTemplate, ...mainList.querySelectorAll('[sw-child-template="main"]')];
                                                    const secondaryItems = [secondaryTemplate, ...secondaryList.querySelectorAll('[sw-child-template="secondary"]')];

                                                    mainItems.forEach( (main, index) => {
                                                        const newIndex = index + 1;
                                                        const title = main.querySelector('[sw-child-title]');
                                                        if (title)
                                                            title.innerText = `Child ${newIndex}`;
                                                        main.querySelectorAll('input, select, textarea').forEach(input => updateNaming(input, newIndex));
                                                        const deleteBtn = main.querySelector('[sw-child-delete]');
                                                        if (deleteBtn)
                                                            deleteBtn.classList.toggle('hide', newIndex === 1);
                                                    }
                                                    );

                                                    secondaryItems.forEach( (secondary, index) => {
                                                        const newIndex = index + 1;
                                                        const title = secondary.querySelector('[sw-child-title]');
                                                        if (title)
                                                            title.innerText = `Child ${newIndex}`;
                                                        secondary.querySelectorAll('input, select, textarea').forEach(input => updateNaming(input, newIndex));

        <script>
            (function() {
                var BASE_URL = 'https://web-leads.replit.app';

                // Form name → which webhook endpoint to use
                // Lead forms go to the routing engine, everything else goes to Riley
                var LEAD_FORMS = ['Initial Request Info Step Form', 'Request Info Form', 'Camp Experts Test Form'];
                var SKIP_FORMS = ['Filter', 'Filters'];

                // 1. Capture & persist UTM params across page navigations
                var params = new URLSearchParams(window.location.search);
                var utmKeys = ['utm_source', 'utm_medium', 'utm_campaign', 'utm_term', 'utm_content'];
                utmKeys.forEach(function(key) {
                    var val = params.get(key);
                    if (val)
                        sessionStorage.setItem('ce_' + key, val);
                });
                if (document.referrer && !sessionStorage.getItem('ce_referrer')) {
                    sessionStorage.setItem('ce_referrer', document.referrer);
                }

                // 2. Intercept form submissions
                document.addEventListener('submit', function(e) {
                    var form = e.target;
                    if (!form || form.tagName !== 'FORM')
                        return;

                    var formName = form.getAttribute('data-name') || form.getAttribute('name') || '';

                    // Skip non-submission forms (filters, etc.)
                    if (SKIP_FORMS.indexOf(formName) !== -1)
                        return;

                    // Collect all form field values
                    var payload = {};
                    var formData = new FormData(form);
                    formData.forEach(function(value, key) {
                        if (value && String(value).trim()) {
                            payload[key] = String(value).trim();
                        }
                    });

                    // Skip if no email (not a real submission)
                    if (!payload.email && !payload.Email && !payload['Email 2'])
                        return;

                    // Add attribution data
                    payload.source_url = window.location.href;
                    utmKeys.forEach(function(key) {
                        var val = sessionStorage.getItem('ce_' + key);
                        if (val)
                            payload[key] = val;
                    });
                    var ref = sessionStorage.getItem('ce_referrer');
                    if (ref)
                        payload.referrer = ref;
                    if (formName)
                        payload._formName = formName;

                    // Route to the right endpoint based on form name
                    var endpoint = LEAD_FORMS.indexOf(formName) !== -1 ? BASE_URL + '/api/webhook/webflow-lead' : BASE_URL + '/api/webhook/webflow-internal';

                    // Fire webhook (non-blocking)
                    if (navigator.sendBeacon) {
                        navigator.sendBeacon(endpoint, new Blob([JSON.stringify(payload)],{
                            type: 'application/json'
                        }));
                    } else {
                        fetch(endpoint, {
                            method: 'POST',
                            headers: {
                                'Content-Type': 'application/json'
                            },
                            body: JSON.stringify(payload),
                            keepalive: true
                        }).catch(function() {});
                    }
                }, true);
            }
            )();
        </script>
